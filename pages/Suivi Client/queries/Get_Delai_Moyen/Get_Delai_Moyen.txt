WITH categories_ref AS (
    SELECT '2T' AS cat, 4 AS delai_defaut
    UNION ALL SELECT '4T', 7
    UNION ALL SELECT 'ROBOT', 5
),
calcul_delais AS (
    SELECT 
        CASE 
            WHEN UPPER(TRIM(COALESCE(i."Categorie_Materiel", pm."Categorie_Materiel", i."Type_Machine", ''))) LIKE '%2T%' THEN '2T'
            WHEN UPPER(TRIM(COALESCE(i."Categorie_Materiel", pm."Categorie_Materiel", i."Type_Machine", ''))) LIKE '%ROBOT%' THEN 'ROBOT'
            ELSE '4T'
        END AS cat,
        EXTRACT(EPOCH FROM (i."Date_Passage_En_Cours" - COALESCE(i."Date_Entree"::timestamptz, i."Date_Heure_Creation")))/86400.0 AS duree_jours
    FROM public."Interventions" i
    LEFT JOIN public."Parc_Materiel" pm ON pm."ID_Machine" = i."ID_Machine"
    WHERE i."Date_Passage_En_Cours" IS NOT NULL
      -- Fenêtre glissante de 45 jours : colle à la saison active (avril prend mars-avril, jamais novembre)
      AND i."Date_Passage_En_Cours" >= CURRENT_DATE - INTERVAL '45 days'
      AND i."Date_Passage_En_Cours" >= COALESCE(i."Date_Entree"::timestamptz, i."Date_Heure_Creation")
)
SELECT 
    r.cat AS "Categorie_Materiel",
    COALESCE(ROUND(AVG(c.duree_jours)), r.delai_defaut) AS "Delai_Moyen_Jours"
FROM categories_ref r
LEFT JOIN calcul_delais c ON c.cat = r.cat
GROUP BY r.cat, r.delai_defaut;
